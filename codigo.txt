import numpy as np

T_FINAL = 60.0
Y0 = (1.0, 0.0, 0.0)
OMEGA = 2 * np.pi / 12

NOMINAL = {
    "Tmax": 4.0,
    "Roce": 0.30,
    "alpha": 0.05,
    "beta": 0.20,
    "gamma": 0.10,
    "delta": 0.40,
    "sigma": 0.70,
    "A": 0.45,
}


def con_parametros(**cambios):
    params = dict(NOMINAL)
    params.update(cambios)
    return params


def R(t, params):
    return params["Roce"] + params["A"] * np.cos(OMEGA * t - np.pi)


def f(t, y, params):
    T, H, P = y
    dT = R(t, params) * T * (1 - T / params["Tmax"]) - params["alpha"] * H
    dH = params["beta"] * T**2 - params["gamma"] * H * P
    dP = params["delta"] * H**2 - params["sigma"] * P
    return np.array([dT, dH, dP])


def euler(params, n_pasos, guardar_cada=1):
    h = T_FINAL / n_pasos
    forma = np.broadcast(*params.values()).shape
    y = np.empty((3,) + forma)
    y[0], y[1], y[2] = Y0

    assert n_pasos % guardar_cada == 0
    n_guardados = n_pasos // guardar_cada
    t_guardado = np.linspace(0, T_FINAL, n_guardados + 1)
    y_guardado = np.empty((n_guardados + 1,) + y.shape)
    y_guardado[0] = y

    with np.errstate(over="ignore", invalid="ignore"):
        for n in range(n_pasos):
            y = y + h * f(n * h, y, params)
            if (n + 1) % guardar_cada == 0:
                y_guardado[(n + 1) // guardar_cada] = y
    return t_guardado, y_guardado

import numpy as np

T_CICLO = 48.0
PERIODO = 12.0


def integral_trapecios(t, y):
    return np.sum((y[1:] + y[:-1]) * np.diff(t).reshape((-1,) + (1,) * (y.ndim - 1)), axis=0) / 2


def indicadores(t, y):
    ciclo = t >= T_CICLO - 1e-9
    tc = t[ciclo]
    T = y[ciclo, 0]
    P = y[ciclo, 2]

    Pmax = P.max(axis=0)
    Pmin = P.min(axis=0)
    Pmed = integral_trapecios(tc, P) / (tc[-1] - tc[0])

    t_Tmax = tc[T.argmax(axis=0)]
    t_Pmax = tc[P.argmax(axis=0)]
    desfase = (t_Pmax - t_Tmax) % PERIODO

    return {
        "SP": (Pmax - Pmin) / Pmed,
        "desfase": desfase,
        "Pmax": Pmax,
        "Pmin": Pmin,
        "Pmed": Pmed,
        "t_Tmax": t_Tmax,
        "t_Pmax": t_Pmax,
    }


def error_periodicidad(t, y):
    k = int(round(PERIODO / (t[1] - t[0])))
    ultimo = y[-k - 1:]
    anterior = y[-2 * k - 1:-k]
    amplitud = ultimo.max(axis=0) - ultimo.min(axis=0)
    diferencia = np.abs(ultimo - anterior).max(axis=0)
    return (diferencia / amplitud).max(axis=0)

import csv
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt

RESULTADOS = Path(__file__).resolve().parent.parent / "results"
RESULTADOS.mkdir(exist_ok=True)

plt.rcParams.update({"figure.dpi": 150, "savefig.bbox": "tight", "axes.grid": True,
                     "grid.alpha": 0.3, "font.size": 9})


def guardar_figura(fig, nombre):
    fig.savefig(RESULTADOS / nombre)
    plt.close(fig)


def cientifica(x):
    mantisa, exponente = f"{x:.2e}".split("e")
    return rf"${mantisa}\cdot10^{{{int(exponente)}}}$"


def guardar_tabla(nombre, encabezados, filas, encabezados_tex=None, filas_tex=None):
    with open(RESULTADOS / f"{nombre}.csv", "w", newline="") as archivo:
        escritor = csv.writer(archivo)
        escritor.writerow(encabezados)
        escritor.writerows(filas)

    encabezados_tex = encabezados_tex or encabezados
    filas_tex = filas_tex or filas
    lineas = [r"\begin{tabular}{" + "l" + "r" * (len(encabezados_tex) - 1) + "}", r"\toprule",
              " & ".join(encabezados_tex) + r" \\", r"\midrule"]
    lineas += [" & ".join(str(celda) for celda in fila) + r" \\" for fila in filas_tex]
    lineas += [r"\bottomrule", r"\end{tabular}"]
    (RESULTADOS / f"{nombre}.tex").write_text("\n".join(lineas) + "\n")

import json

import matplotlib.pyplot as plt
import numpy as np

from modelo import NOMINAL, T_FINAL, euler
from salida import RESULTADOS, cientifica, guardar_figura, guardar_tabla

PUNTOS_POR_MES = 128
K_REF = 14
PASOS_DUPLICADOS = [2.0**k for k in range(-7, 3)]
PASOS_INTERMEDIOS = [0.75, 1.25, 1.5]


def referencia(k):
    n_pasos = int(T_FINAL) * 2**k
    return euler(NOMINAL, n_pasos, guardar_cada=2**k // PUNTOS_POR_MES)


def errores_maximos(h, t_ref, y_ref):
    t, y = euler(NOMINAL, round(T_FINAL / h))
    salto = round(h * PUNTOS_POR_MES)
    assert np.allclose(t, t_ref[::salto])
    if not np.isfinite(y).all():
        return np.full(3, np.inf), t, y
    return np.abs(y - y_ref[::salto]).max(axis=0), t, y


def main():
    resumen = {}

    t_ref, y_ref = referencia(K_REF)
    _, y_fina = referencia(K_REF + 1)
    diferencia = np.abs(y_ref - y_fina).max(axis=0)
    resumen["h_ref"] = 2.0**-K_REF
    resumen["dif_ref_vs_fina_T"] = diferencia[0]
    resumen["error_estimado_ref_T"] = 2 * diferencia[0]

    resultados = {}
    for h in sorted(PASOS_DUPLICADOS + PASOS_INTERMEDIOS):
        resultados[h] = errores_maximos(h, t_ref, y_ref)

    filas, filas_tex = [], []
    for h in PASOS_DUPLICADOS:
        err = resultados[h][0]
        anterior = resultados.get(h / 2)
        if np.isfinite(err[0]) and anterior is not None:
            cociente = err[0] / anterior[0][0]
            orden = np.log(cociente) / np.log(2)
            cociente, orden = f"{cociente:.3f}", f"{orden:.3f}"
        else:
            cociente = orden = "--"
        if np.isfinite(err[0]):
            filas.append([h, round(T_FINAL / h), *(f"{e:.4e}" for e in err), cociente, orden])
            filas_tex.append([f"{h:g}", round(T_FINAL / h), *(cientifica(e) for e in err), cociente, orden])
        else:
            filas.append([h, round(T_FINAL / h), "inf", "inf", "inf", cociente, orden])
            filas_tex.append([f"{h:g}", round(T_FINAL / h), *["diverge"] * 3, cociente, orden])
    guardar_tabla(
        "tabla_convergencia",
        ["h", "N", "Err_T", "Err_H", "Err_P", "Err_T(h)/Err_T(h/2)", "p"], filas,
        ["$h$ [mes]", "$N$", r"$\mathrm{Err}_T$ [°C]", r"$\mathrm{Err}_H$ [u.h.]",
         r"$\mathrm{Err}_P$ [u.p.]", r"$\frac{\mathrm{Err}_T(h)}{\mathrm{Err}_T(h/2)}$", "$p$"],
        filas_tex)
    resumen["error_T_menor_paso"] = resultados[PASOS_DUPLICADOS[0]][0][0]
    resumen["relacion_ref_menor_paso"] = resumen["error_estimado_ref_T"] / resumen["error_T_menor_paso"]

    fig, ax = plt.subplots(figsize=(5.5, 3.8))
    hs = np.array([h for h in sorted(resultados) if np.isfinite(resultados[h][0][0])])
    for i, (nombre, marca) in enumerate([("T [°C]", "o-"), ("H [u.h.]", "s--"), ("P [u.p.]", "^--")]):
        ax.loglog(hs, [resultados[h][0][i] for h in hs], marca, ms=4, lw=1, label=f"Err de {nombre}")
    ax.loglog(hs, resultados[hs[0]][0][0] * hs / hs[0], "k:", label="pendiente 1")
    divergentes = [h for h in sorted(resultados) if not np.isfinite(resultados[h][0][0])]
    for h in divergentes:
        ax.axvline(h, color="r", lw=0.8, ls="-.")
    ax.axvline(divergentes[0], color="r", lw=0.8, ls="-.", label="diverge (overflow)")
    ax.set_xlabel("h [mes]")
    ax.set_ylabel("máximo error absoluto")
    ax.set_title("Error de Euler respecto de la referencia")
    ax.legend(fontsize=7)
    guardar_figura(fig, "fig_convergencia.png")

    ultimo_ciclo_ref = t_ref >= 48
    filas, filas_tex = [], []
    for h in [0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 4.0]:
        err, t, y = resultados[h]
        if np.isfinite(err[0]):
            P_ciclo = y[t >= 48, 2]
            fila = [f"{h:g}", f"{err[0]:.3f}", f"{err[2]:.2f}", f"{P_ciclo.min():.2f}", f"{P_ciclo.max():.2f}"]
        else:
            with np.errstate(invalid="ignore"):
                t_explota = t[np.argmax(~(np.abs(y[:, 0]) < 100))]
            fila = [f"{h:g}", "diverge", "diverge", "--", f"$|T|>100$ en $t={t_explota:g}$"]
        filas.append(fila)
    P_ref = y_ref[ultimo_ciclo_ref, 2]
    filas.insert(0, ["ref.", "0", "0", f"{P_ref.min():.2f}", f"{P_ref.max():.2f}"])
    guardar_tabla(
        "tabla_estabilidad", ["h", "Err_T", "Err_P", "P_min ultimo ciclo", "P_max ultimo ciclo"], filas,
        ["$h$ [mes]", r"$\mathrm{Err}_T$ [°C]", r"$\mathrm{Err}_P$ [u.p.]",
         r"$P_{\min}$ en $[48,60]$", r"$P_{\max}$ en $[48,60]$"])

    fig, ejes = plt.subplots(2, 2, figsize=(8, 5), sharex=True)
    for ax, h in zip(ejes.flat, [0.5, 1.0, 1.25, 1.5]):
        _, t, y = resultados[h]
        ax.plot(t_ref, y_ref[:, 2], "k-", lw=1, label="referencia")
        ax.plot(t, y[:, 2], "r.-", lw=0.7, ms=3, label=f"Euler h = {h:g}")
        ax.set_ylim(-2, 25)
        ax.set_title(f"h = {h:g} mes")
        ax.legend(fontsize=7, loc="upper left")
    for ax in ejes[1]:
        ax.set_xlabel("t [mes]")
    for ax in ejes[:, 0]:
        ax.set_ylabel("P [u.p.]")
    fig.suptitle("P(t) con pasos cercanos al límite de estabilidad")
    guardar_figura(fig, "fig_inestabilidad.png")

    (RESULTADOS / "parte1_resumen.json").write_text(json.dumps(resumen, indent=2))
    print(json.dumps(resumen, indent=2))


if __name__ == "__main__":
    main()

import json

N_PASOS = 7680

RANGOS = {"A": (0.25, 0.60), "beta": (0.10, 0.30), "gamma": (0.05, 0.20), "sigma": (0.40, 1.00)}
SIMBOLO = {"A": "A", "beta": r"\beta", "gamma": r"\gamma", "sigma": r"\sigma"}
UNIDAD = {"A": "mes$^{-1}$", "beta": "u.h. mes$^{-1}$ °C$^{-2}$",
          "gamma": "mes$^{-1}$ u.p.$^{-1}$", "sigma": "mes$^{-1}$"}

BARRIDO = {"A": 15, "beta": 21, "gamma": 16, "sigma": 13}
GRILLA = {"A": 8, "beta": 9, "gamma": 7, "sigma": 7}

COTA = 100.0
TOL_PERIODICIDAD = 0.01


def valores(nombre, cantidad):
    return np.round(np.linspace(*RANGOS[nombre], cantidad), 6)


def evaluar(params, n_pasos=N_PASOS):
    t, y = euler(params, n_pasos)
    with np.errstate(invalid="ignore"):
        res = indicadores(t, y)
        res["periodicidad"] = error_periodicidad(t, y)
        res["maximo"] = np.abs(y).max(axis=(0, 1))
    ciclo = t >= T_CICLO
    res["Tmin"] = y[ciclo, 0].min(axis=0)
    res["Tmax"] = y[ciclo, 0].max(axis=0)
    res["Hmed"] = integral_trapecios(t[ciclo], y[ciclo, 1]) / (t[-1] - T_CICLO)
    return res


def es_valido(res):
    with np.errstate(invalid="ignore"):
        return (res["maximo"] < COTA) & (res["periodicidad"] < TOL_PERIODICIDAD)


def buscar_en_grilla(ejes):
    filas = []
    B, G, S = (m.ravel() for m in np.meshgrid(ejes["beta"], ejes["gamma"], ejes["sigma"], indexing="ij"))
    for A in ejes["A"]:
        params = con_parametros(A=A, beta=B, gamma=G, sigma=S)
        res = evaluar(params)
        valido = es_valido(res)
        filas += zip(np.full(B.size, A), B, G, S, res["SP"], res["desfase"],
                     res["periodicidad"], res["maximo"], res["t_Pmax"] > res["t_Tmax"], valido)
    return np.array(filas)

def maximizar(resumen):
    ejes = {nombre: valores(nombre, GRILLA[nombre]) for nombre in RANGOS}
    grilla = buscar_en_grilla(ejes)
    columnas = ["A", "beta", "gamma", "sigma", "SP", "desfase", "periodicidad", "maximo",
                "maxP_despues_de_maxT", "valido"]
    np.savetxt(RESULTADOS / "grilla.csv", grilla, delimiter=",", header=",".join(columnas), comments="", fmt="%.6g")

    validas = grilla[grilla[:, -1] == 1]
    orden = np.argsort(validas[:, 4])[::-1]
    mejor = validas[orden[0]]
    guardar_tabla("tabla_grilla_mejores", columnas[:7],
                  [[f"{f[0]:.2f}", f"{f[1]:.3f}", f"{f[2]:.3f}", f"{f[3]:.2f}", f"{f[4]:.4f}", f"{f[5]:.3f}",
                    f"{100 * f[6]:.2f}"] for f in validas[orden[:8]]],
                  ["$A$", r"$\beta$", r"$\gamma$", r"$\sigma$", "$S_P$", r"$\Delta t_{TP}$ [mes]", "Periodicidad [\\%]"])

    SP = grilla[:, 4].reshape([GRILLA[n] for n in RANGOS])
    resumen["grilla"] = {
        "combinaciones": len(grilla),
        "validas": len(validas),
        "peor_periodicidad": float(grilla[:, 6].max()),
        "mayor_valor_absoluto": float(grilla[:, 7].max()),

        "desfase_min": float(validas[:, 5].min()),
        "desfase_max": float(validas[:, 5].max()),
        "casos_maxP_antes_de_maxT": int((grilla[:, 8] == 0).sum()),
        "mejor": dict(zip(columnas[:7], map(float, mejor[:7]))),
        "SP_creciente_en": {n: bool((np.diff(SP, axis=i) > 0).all()) for i, n in enumerate(RANGOS)},
        "SP_decreciente_en": {n: bool((np.diff(SP, axis=i) < 0).all()) for i, n in enumerate(RANGOS)},
    }
    return dict(zip(RANGOS, mejor[:4]))

                 "Tmax": r"$T_{\max}$ [°C]", "Hmed": r"$H_{med}$ [u.h.]", "periodicidad": "Periodicidad [\\%]"}
    for clave, etiqueta in etiquetas.items():
        factor = 100 if clave == "periodicidad" else 1
        filas.append([etiqueta] + [f"{factor * r[clave]:.3f}" for r in resultados.values()])
    guardar_tabla("tabla_comparacion", ["", *escenarios], filas)
    resumen["escenarios"] = {nombre.replace("$", ""): {k: float(v) for k, v in r.items()}
                             for nombre, r in resultados.items()}

    nombres = ["T [°C]", "H [u.h.]", "P [u.p.]"]
    fig, ejes = plt.subplots(3, 2, figsize=(10, 6), sharex="col", sharey="row",
                             gridspec_kw={"width_ratios": [2, 1]})
    for nombre, estilo in zip(["Nominal", "Mayor $S_P$"], ["C0-", "C3-"]):
        t, y = euler(con_parametros(**escenarios[nombre]), N_PASOS)
        ciclo = t >= T_CICLO
        for i in range(3):
            ejes[i, 0].plot(t, y[:, i], estilo, lw=1, label=nombre)
            ejes[i, 1].plot(t[ciclo], y[ciclo, i], estilo, lw=1)
            ejes[i, 0].set_ylabel(nombres[i])
    ejes[0, 0].legend(fontsize=7)
    ejes[0, 0].set_title("Simulación completa")
    ejes[0, 1].set_title("Último ciclo anual")
    ejes[2, 0].set_xlabel("t [mes]")
    ejes[2, 1].set_xlabel("t [mes]")
    ejes[2, 0].set_xticks(range(0, 61, 6))
    ejes[2, 1].set_xticks(range(48, 61, 2))
    guardar_figura(fig, "fig_comparacion.png")
